/** Only surrounding UI methods are boundaries. The async methods below are extracted verbatim from production on every run. */
class ExtractedCallbacks(val launcher: Launcher) : BgDataModel.Callbacks {
    // ACTIVE_FIELD
    var synchronouslyBoundPages = IntSet()
    var pagesToBindSynchronously = IntSet()
    private val TAG = "ModelCallbacks"
    private val DISPLAY_WORKSPACE_TRACE_METHOD_NAME = "workspace"
    private val SINGLE_TRACE_COOKIE = 0
    fun useModelRepositoryBinding() = false
    fun clearPendingBinds() { launcher.clears++ }
    fun getPagesToBindSynchronously(screens: IntArray) = IntSet.wrap(0)
    fun startBinding() { launcher.visible.clear() }
    fun bindScreens(screens: IntArray) {}
    fun finishBindingItems(pages: IntSet) { launcher.finished++ }
    fun currentScreenContentFilter(pages: IntSet) = Predicate<ItemInfo> { it.screenId == 0 }
    fun onInitialBindComplete(pages: IntSet, pending: RunnableList, complete: RunnableList, count: Int, sync: Boolean) {
        launcher.pending.add(pending)
        complete.add { pending.executeAllAndDestroy() }
        launcher.bindComplete(count, sync)
    }
    fun bindRepositorySnapshot(data: WorkspaceData) = bindModelWithAsyncInflation(data, false, "repository")
    // PRODUCTION_METHODS
}
